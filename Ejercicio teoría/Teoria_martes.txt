let vprod v1 v2 =
	if Array.length v1 = Array.length v2 then
		let p = ref 0 in 
		for i = 0 to Array.length v1 do
		 p := !p + v1.(i) * v2.(i)
		done;
		!p
	else
		raise (Invalid_argument "vprod")

type 'a ref = {mutable contents : 'a}

let ref x = {contents = x}

(* (!) -> obtener contenido de una variable *)

let (!) v = v.contents;;

(* (:=) -> cambiar el valor a una variable *)

let (:=) v x = v.contents <- x

module Counter () : sig
	val next : unit -> int
	val reset : unit -> unit
end =
struct
	let n = ref 0
	let next () = n := in + 1; in
	let reset () = n := 0
end
